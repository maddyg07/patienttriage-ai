# PatientTriage.ai -- Round 2 prototype
# Phase 1 needs nothing beyond the Python standard library (3.10+).
# The lines below are commented out until the phase that introduces them.

# streamlit>=1.32      # Phase 12 -- dashboard

# pytest>=8.0          # Phase 15 -- NOT required. The suite is plain
#                      # unittest from the standard library, so
#                      # `python -m scripts.run_tests` works on a clean
#                      # machine with no network. pytest discovers and runs
#                      # tests/ unchanged if you prefer it.

# Phase 17 -- live intake with camera, microphone and speech.
# Still nothing to install. The capture layer is the browser's own
# getUserMedia, Web Audio and SpeechRecognition APIs; the server is
# http.server from the standard library. Speech recognition needs
# Chrome or Edge; every other channel works in any browser.

# Phase 18/19 -- real-time clinic, two dashboards, model-backed language layer.
# Still nothing to install. Server-Sent Events and ThreadingHTTPServer are
# standard library; every model call is one urllib POST.
#
# Set ONE of these for real natural-language understanding. Whichever key you
# have is the one that gets used, checked in this order:
#
#   export OPENAI_API_KEY=sk-...            # gpt-4o-mini by default
#   export GOOGLE_API_KEY=...               # gemini-2.0-flash by default
#   export ANTHROPIC_API_KEY=sk-ant-...     # claude-sonnet-4-6 by default
#
# Optional:
#   export PATIENTTRIAGE_MODEL=<model id>   # override the default model
#   export PATIENTTRIAGE_PROVIDER=openai    # force one vendor
#   export PATIENTTRIAGE_PROVIDER=local     # force the offline matcher
#   export OPENAI_BASE_URL=...              # any OpenAI-compatible endpoint
#
# With no key the deterministic matcher serves, both dashboards show DEGRADED,
# and the rule-based emergency gate is unaffected.
