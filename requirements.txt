# Phase 17 -- live intake with camera, microphone and speech.
# Still nothing to install. The capture layer is the browser's own
# getUserMedia, Web Audio and SpeechRecognition APIs; the server is
# http.server from the standard library. Speech recognition needs
# Chrome or Edge; every other channel works in any browser.

# Phase 18-20 -- real-time clinic, two dashboards, model-backed language layer.
# Still nothing to install. Server-Sent Events and ThreadingHTTPServer are
# standard library; every model call is one urllib POST.
#
# Set ONE of these. Whichever key is present is used, tried in this order.
# The two free tiers come first on purpose: an account with no credit balance
# fails exactly like no key at all, except that it looks configured.
#
#   export GOOGLE_API_KEY=...        # FREE, no card    aistudio.google.com/apikey
#   export GROQ_API_KEY=gsk_...      # FREE, no card    console.groq.com/keys
#   export OPENAI_API_KEY=sk-...     # paid, needs credit
#   export ANTHROPIC_API_KEY=sk-...  # paid, needs credit
#
# Optional:
#   export PATIENTTRIAGE_MODEL=<id>       # override the default model
#   export PATIENTTRIAGE_PROVIDER=gemini  # force one vendor
#   export PATIENTTRIAGE_PROVIDER=local   # force the offline matcher
#   export OPENAI_BASE_URL=...            # any OpenAI-compatible endpoint
#   export GROQ_BASE_URL=...
#
# Free-tier model ids move. If a call returns 404, set PATIENTTRIAGE_MODEL from
# ai.google.dev/gemini-api/docs/models or console.groq.com/docs/models.
#
# With no key the deterministic matcher serves, both dashboards show DEGRADED
# with the reason, and the rule-based emergency gate is unaffected.
